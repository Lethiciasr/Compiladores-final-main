main() {
    int grade[3][3];
    int linha;
    int coluna;
    int valor;

    valor = 1;

    for (linha = 0; linha < 3; linha++) {
        for (coluna = 0; coluna < 3; coluna++) {
            grade[linha][coluna] = valor;
            valor++;
        }
    }

    print(grade[0][0]);
    print(grade[1][1]);
    print(grade[2][2]);
}
